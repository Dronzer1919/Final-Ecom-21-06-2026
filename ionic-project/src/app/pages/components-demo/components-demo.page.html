1. Import Components

import {{ '{' }}
  ThemedButtonComponent,
  ThemedCardComponent,
  ThemedAccordionComponent,
  ThemedTableComponent
{{ '}' }} from './components/reusable';

2. Add to Component Imports

@Component({{ '{' }}
  imports: [
    ThemedButtonComponent,
    ThemedCardComponent,
    // ... other imports
  ]
{{ '}' }})

3. Use in Templates

<app-themed-button color="primary">
  Click Me
</app-themed-button>

<app-themed-card title="My Card">
  Card content here
</app-themed-card>
